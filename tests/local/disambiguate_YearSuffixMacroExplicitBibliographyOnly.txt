>>===== MODE =====>>
citation
<<===== MODE =====<<


>>===== DESCRIPTION =====>>
An explicit year-suffix element in a bibliography-only macro must not disable
implicit year-suffix injection in citations.
<<===== DESCRIPTION =====<<


>>===== RESULT =====>>
Smith 1980a; Smith 1980b
<<===== RESULT =====<<


>>===== CSL =====>>
<style xmlns="http://purl.org/net/xbiblio/csl" version="1.0">
  <info>
    <id>test-year-suffix-macro-explicit-bibliography-only</id>
    <title>Year suffix in bibliography macro only</title>
    <updated>2026-11-18T00:00:00+00:00</updated>
  </info>
  <macro name="year-suffix">
    <text variable="year-suffix"/>
  </macro>
  <citation disambiguate-add-year-suffix="true">
    <layout delimiter="; ">
      <names variable="author"><name form="short"/></names>
      <date variable="issued" prefix=" ">
        <date-part name="year"/>
      </date>
    </layout>
  </citation>
  <bibliography>
    <layout>
      <names variable="author"><name form="short"/></names>
      <date variable="issued" prefix=" ">
        <date-part name="year"/>
      </date>
      <text macro="year-suffix"/>
    </layout>
  </bibliography>
</style>
<<===== CSL =====<<

[

    "id": "a",
    "type": "book",
    "author": [{"family": "Smith"}],
    "issued": {"date-parts": [[1980]]}
  },
  {
    "id": "b",
    "type": "book",
    "author": [{"family": "Smith"}],
    "issued": {"date-parts": [[1980]]}
  }
]
